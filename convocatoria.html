-- NEXIA Convocatorias: adjudicación atómica "primero en tomar" CREATE OR REPLACE FUNCTION public.nexi_tomar_convocatoria( p_match_id uuid ) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER AS $$ DECLARE v_match record; v_tomado_id uuid; BEGIN SELECT id, solicitud_id, interesado_id, estado INTO v_match FROM public.nexi_convocatorias_matches WHERE id = p_match_id FOR UPDATE; IF NOT FOUND THEN RETURN jsonb_build_object( 'ok', false, 'motivo', 'match_no_encontrado' ); END IF; IF v_match.estado = 'tomada' THEN RETURN jsonb_build_object( 'ok', true, 'solicitud_id', v_match.solicitud_id, 'match_id', v_match.id, 'interesado_id', v_match.interesado_id, 'motivo', 'ya_tomada_por_este_interesado' ); END IF; IF v_match.estado <> 'pendiente' THEN RETURN jsonb_build_object( 'ok', false, 'solicitud_id', v_match.solicitud_id, 'motivo', 'no_disponible' ); END IF; PERFORM 1 FROM public.nexi_convocatorias_matches WHERE solicitud_id = v_match.solicitud_id FOR UPDATE; SELECT id INTO v_tomado_id FROM public.nexi_convocatorias_matches WHERE solicitud_id = v_match.solicitud_id AND estado = 'tomada' AND id <> p_match_id LIMIT 1; IF v_tomado_id IS NOT NULL THEN UPDATE public.nexi_convocatorias_matches SET estado = 'cerrada', updated_at = now() WHERE id = p_match_id AND estado = 'pendiente'; RETURN jsonb_build_object( 'ok', false, 'solicitud_id', v_match.solicitud_id, 'motivo', 'tomada_por_otro' ); END IF; UPDATE public.nexi_convocatorias_matches SET estado = 'tomada', tomado_at = now(), updated_at = now() WHERE id = p_match_id; UPDATE public.nexi_convocatorias_matches SET estado = 'cerrada', updated_at = now() WHERE solicitud_id = v_match.solicitud_id AND id <> p_match_id AND estado = 'pendiente'; RETURN jsonb_build_object( 'ok', true, 'solicitud_id', v_match.solicitud_id, 'match_id', p_match_id, 'interesado_id', v_match.interesado_id, 'motivo', 'tomada' ); END; $$;
